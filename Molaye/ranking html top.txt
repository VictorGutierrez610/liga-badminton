const TIPOS_BASE = ['IM', 'IF', 'DM', 'DF', 'DX'];
const NOMBRE_TIPO = {
  IM: 'Individual masculino', IF: 'Individual femenino',
  DM: 'Dobles masculino', DF: 'Dobles femenino', DX: 'Dobles mixto'
};
// En dobles no hay niveles A/B: solo se elige el nivel en individual.
function usaNivel(tipo) { return tipo === 'IM' || tipo === 'IF'; }

let DATA = null;
let TEMPORADA = '';
let NIVEL = 'A';
let TIPO = 'IM';

function $(id) { return document.getElementById(id); }
function esc(v) {
  return String(v === undefined || v === null ? '' : v)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
function normaliza(v) { return String(v || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase(); }
function fechaCorta(iso) {
  const p = String(iso).split('-');
  return p.length === 3 ? p[2] + '/' + p[1] + '/' + p[0].slice(2) : iso;
}
function signo(n) { return n > 0 ? '+' + n : String(n); }

function cargar() {
  $('fallo').style.display = 'none';
  $('btnAct').disabled = true;
  if (!DATA) { $('cargando').style.display = 'block'; }
  google.script.run
    .withSuccessHandler(function(datos) {
      $('cargando').style.display = 'none';
      $('btnAct').disabled = false;
      DATA = datos;
      iniciarVista();
    })
    .withFailureHandler(function(error) {
      $('cargando').style.display = 'none';
      $('btnAct').disabled = false;
      $('falloTxt').textContent = 'No se han podido cargar los rankings.\n' + String((error && error.message) || error).replace(/^Error:\s*/, '');
      $('fallo').style.display = 'block';
    })
    .getRanking();
}

function iniciarVista() {
  $('generado').textContent = 'Actualizado: ' + DATA.generado;
  $('contenido').style.display = 'block';

  const temporadas = DATA.temporadas || [];
  const sel = $('selTemporada');
  sel.innerHTML = temporadas.map(t => '<option value="' + esc(t.id) + '">Temporada ' + esc(t.etiqueta) + '</option>').join('');
  sel.style.display = temporadas.length > 1 ? '' : 'none';
  if (!temporadas.some(t => t.id === TEMPORADA)) TEMPORADA = temporadas.length ? temporadas[0].id : '';
  sel.value = TEMPORADA;
  pintarControles();
  pintarTabla();
}

function temporadaActual() { return (DATA.temporadas || []).find(t => t.id === TEMPORADA) || null; }
function cambiarTemporada() { TEMPORADA = $('selTemporada').value; pintarControles(); pintarTabla(); }

function pintarControles() {
  const T = temporadaActual();
  const cats = T ? Object.keys(T.categorias).map(c => T.categorias[c]) : [];

  // Niveles: A y B siempre; si en la hoja hubiera otros (por ejemplo categorías antiguas sin nivel) se añaden.
  const niveles = ['A', 'B'];
  cats.forEach(c => { if (usaNivel(c.tipo) && niveles.indexOf(c.nivel) === -1) niveles.push(c.nivel); });
  if (niveles.indexOf(NIVEL) === -1) NIVEL = 'A';
  const tipos = TIPOS_BASE.slice();
  cats.forEach(c => { if (tipos.indexOf(c.tipo) === -1) tipos.push(c.tipo); });
  if (tipos.indexOf(TIPO) === -1) TIPO = 'IM';

  $('segNivel').innerHTML = '';
  $('segNivel').style.display = usaNivel(TIPO) ? '' : 'none';
  niveles.forEach(n => {
    const b = document.createElement('button');
    b.textContent = n === '' ? 'Sin nivel' : (n.length === 1 ? 'Nivel ' + n : n);
    if (n === NIVEL) b.classList.add('active');
    b.onclick = function() { NIVEL = n; pintarControles(); pintarTabla(); };
    $('segNivel').appendChild(b);
  });

  $('tabsTipo').innerHTML = '';
  tipos.forEach(t => {
    const b = document.createElement('button');
    b.innerHTML = esc(t) + '<small>' + esc(NOMBRE_TIPO[t] || '') + '</small>';
    if (t === TIPO) b.classList.add('active');
    b.onclick = function() { TIPO = t; pintarControles(); pintarTabla(); };
    $('tabsTipo').appendChild(b);
  });
}

// Índices de las jornadas que cuentan (las N mejores; en empate, las primeras)
function jornadasQueCuentan(pts, n) {
  return pts.map((p, i) => ({ p: p, i: i })).filter(x => x.p !== null)
    .sort((a, b) => (b.p - a.p) || (a.i - b.i)).slice(0, n).map(x => x.i);
}

function pintarTabla() {
  if (!DATA) return;
  const T = temporadaActual();
  const conNivel = usaNivel(TIPO);
  const nivelTxt = NIVEL === '' ? 'Sin nivel' : (NIVEL.length === 1 ? 'Nivel ' + NIVEL : NIVEL);
  $('titulo').textContent = conNivel ? TIPO + ' · ' + nivelTxt : TIPO;
  $('subtitulo').textContent = (NOMBRE_TIPO[TIPO] || TIPO) + (T ? ' · Temporada ' + T.etiqueta : '');
  const zona = $('zonaTabla');
  const N = DATA.mejoresJornadas || 3;

  const clave = (!conNivel || NIVEL === '') ? TIPO : TIPO + '-' + NIVEL;
  const cat = T && T.categorias[clave];
  if (!cat || !cat.jugadores.length) {
    zona.innerHTML = '<div class="empty" style="margin-top:12px">Todavía no hay resultados de ' + esc(clave) + (T ? ' en esta temporada' : '') + '.</div>';
    $('notas').textContent = '';
    $('descargas').style.display = 'none';
    return;
  }

  const q = normaliza($('buscar').value.trim());
  const jugadores = cat.jugadores.filter(j => !q || normaliza(j.nombre).includes(q));
  if (!jugadores.length) {
    zona.innerHTML = '<div class="empty" style="margin-top:12px">Ningún jugador coincide con la búsqueda.</div>';
    return;
  }

  let h = '<div class="table-wrap"><table class="rank"><thead><tr>';
  h += '<th class="c-pos">#</th><th class="c-name">Jugador</th><th class="c-total">Total<small>' + N + ' mejores</small></th>';
  cat.jornadas.forEach((j, i) => { h += '<th class="' + (i === 0 ? 'sep' : '') + '">J' + j.n + '<small>' + esc(fechaCorta(j.fecha)) + '</small></th>'; });
  h += '<th class="sep">PG</th><th>PP</th><th>Dif</th><th class="sep">PF</th><th>PC</th><th>Dif</th>';
  for (let k = 1; k <= N; k++) h += '<th class="' + (k === 1 ? 'sep' : '') + '">Mejor ' + k + '</th>';
  h += '</tr></thead><tbody>';

  jugadores.forEach(j => {
    const cuentan = jornadasQueCuentan(j.pts, N);
    h += '<tr class="' + (j.pos <= 3 ? 'p' + j.pos : '') + '">';
    h += '<td class="c-pos">' + j.pos + '</td><td class="c-name">' + esc(j.nombre) + '</td><td class="c-total">' + j.total + '</td>';
    j.pts.forEach((p, i) => {
      if (p === null) h += '<td class="none' + (i === 0 ? ' sep' : '') + '">–</td>';
      else h += '<td class="' + (cuentan.indexOf(i) !== -1 ? 'cuenta' : 'nocuenta') + (i === 0 ? ' sep' : '') + '">' + p + '</td>';
    });
    h += '<td class="sep">' + j.PG + '</td>';
    h += '<td>' + (j.PP === null ? '–' : j.PP) + '</td>';
    h += '<td class="' + (j.difPartidos < 0 ? 'neg' : '') + '">' + (j.difPartidos === null ? '–' : signo(j.difPartidos)) + '</td>';
    h += '<td class="sep">' + j.PF + '</td><td>' + j.PC + '</td><td class="' + (j.difPuntos < 0 ? 'neg' : '') + '">' + signo(j.difPuntos) + '</td>';
    j.mejores.forEach((p, k) => { h += '<td class="' + (k === 0 ? 'sep ' : '') + (p === null ? 'none' : '') + '">' + (p === null ? '–' : p) + '</td>'; });
    h += '</tr>';
  });
  h += '</tbody></table></div>';
  zona.innerHTML = h;
  $('descargas').style.display = 'flex';

  $('notas').innerHTML =
    'Cada jugador suma los puntos de sus <b>' + N + ' mejores jornadas</b> (en verde en la tabla). ' +
    '<b>PG/PP</b>: partidos ganados/perdidos · <b>PF/PC</b>: puntos a favor/en contra · <b>Dif</b>: diferencia. ' +
    'PG, PP, PF y PC suman todas las jornadas de la temporada. ' +
    '<br>Desempates: 1) victorias, 2) diferencia de partidos, 3) diferencia de puntos, 4) puntos a favor.' +
    (DATA.hayDerrotas ? '' : '<br><span class="err">Aún no se han calculado las derrotas: ejecuta «actualizarResultadosIndividuales» en el proyecto del gestor.</span>');
}

function slug(s) { return String(s).replace(/[^\w-]+/g, '_'); }

function fitTexto(c, txt, maxW, fs, peso) {
  let s = fs;
  do { c.font = peso + ' ' + s + 'px Arial, Helvetica, sans-serif'; if (c.measureText(txt).width <= maxW) break; s -= 1; } while (s > 10);
  return s;
}

// Mete un JPEG en un PDF de una página (pw x ph en puntos), centrado y ajustado.
function crearPdf(jpegUrl, iw, ih, pw, ph) {
  const bin = atob(jpegUrl.split(',')[1]), jpg = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) jpg[i] = bin.charCodeAt(i);
  const k = Math.min(pw / iw, ph / ih), w = (iw * k).toFixed(2), h = (ih * k).toFixed(2);
  const x = ((pw - iw * k) / 2).toFixed(2), y = ((ph - ih * k) / 2).toFixed(2);
  const enc = new TextEncoder(), parts = [], offs = []; let len = 0;
  const add = d => { const u = typeof d === 'string' ? enc.encode(d) : d; parts.push(u); len += u.length; };
  const obj = (n, head, stream) => { offs[n] = len; add(n + ' 0 obj\n' + head + '\n'); if (stream) { add('stream\n'); add(stream); add('\nendstream\n'); } add('endobj\n'); };
  const cont = 'q ' + w + ' 0 0 ' + h + ' ' + x + ' ' + y + ' cm /Im0 Do Q';
  add('%PDF-1.4\n');
  obj(1, '<< /Type /Catalog /Pages 2 0 R >>');
  obj(2, '<< /Type /Pages /Kids [3 0 R] /Count 1 >>');
  obj(3, '<< /Type /Page /Parent 2 0 R /MediaBox [0 0 ' + pw + ' ' + ph + '] /Resources << /XObject << /Im0 4 0 R >> >> /Contents 5 0 R >>');
  obj(4, '<< /Type /XObject /Subtype /Image /Width ' + iw + ' /Height ' + ih + ' /ColorSpace /DeviceRGB /BitsPerComponent 8 /Filter /DCTDecode /Length ' + jpg.length + ' >>', jpg);
  obj(5, '<< /Length ' + cont.length + ' >>', cont);
  const xr = len; add('xref\n0 6\n0000000000 65535 f \n');
  for (let n = 1; n <= 5; n++) add(String(offs[n]).padStart(10, '0') + ' 00000 n \n');
  add('trailer\n<< /Size 6 /Root 1 0 R >>\nstartxref\n' + xr + '\n%%EOF');
  return new Blob(parts, { type: 'application/pdf' });
}

function descargar(blob, nombre) {
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob); a.download = nombre;
  document.body.appendChild(a); a.click();
  setTimeout(function () { URL.revokeObjectURL(a.href); a.remove(); }, 1500);
}

function entregar(cv, formato, nombre, pw, ph) {
  try {
    if (formato === 'jpg') cv.toBlob(function (b) { descargar(b, nombre + '.jpg'); }, 'image/jpeg', 0.95);
    else descargar(crearPdf(cv.toDataURL('image/jpeg', 0.95), cv.width, cv.height, pw, ph), nombre + '.pdf');
  } catch (e) { alert('No se ha podido generar el archivo: ' + e.message); }
}

/* ================================================================
   RANKING -> JPEG / PDF.  datos = lo que devuelve getRanking()
   ================================================================ */
function exportarRanking(datos, temporadaId, cat, formato) {
  const T = (datos.temporadas || []).find(t => t.id === temporadaId), C = T && T.categorias[cat];
  if (!C || !C.jugadores.length) { alert('No hay datos de esa categoría.'); return; }
  const N = datos.mejoresJornadas, sig = v => v === null ? '–' : (v > 0 ? '+' + v : String(v));
  const fc = iso => { const p = String(iso).split('-'); return p.length === 3 ? p[2] + '/' + p[1] + '/' + p[0].slice(2) : iso; };

  // Mismas columnas que la tabla de la web: #, Jugador, Total, J1..Jn, PG PP Dif, PF PC Dif, Mejor 1..N
  const cols = [{ h: '#', w: 64, a: 'center' }, { h: 'Jugador/a', w: 300, a: 'left' }, { h: 'Total', s: N + ' mejores', w: 110, a: 'center', tot: 1 }];
  C.jornadas.forEach((j, i) => cols.push({ h: 'J' + j.n, s: fc(j.fecha), w: 84, a: 'center', sep: i === 0, jor: i }));
  cols.push({ h: 'PG', w: 70, a: 'center', sep: 1 }, { h: 'PP', w: 70, a: 'center' }, { h: 'Dif', w: 76, a: 'center' },
            { h: 'PF', w: 76, a: 'center', sep: 1 }, { h: 'PC', w: 76, a: 'center' }, { h: 'Dif', w: 84, a: 'center' });
  for (let k = 1; k <= N; k++) cols.push({ h: 'Mejor ' + k, w: 100, a: 'center', sep: k === 1 });
  const fila = j => {
    const v = [j.pos, j.nombre, j.total];
    j.pts.forEach(p => v.push(p === null ? '–' : p));
    v.push(j.PG, j.PP === null ? '–' : j.PP, sig(j.difPartidos), j.PF, j.PC, sig(j.difPuntos));
    j.mejores.forEach(m => v.push(m === null ? '–' : m));
    return v;
  };

  const PAD = 40, top = 150, headH = 84, rowH = 56, n = C.jugadores.length;
  const W = cols.reduce((s, k) => s + k.w, 0) + 2 * PAD, H = top + headH + n * rowH + 110;
  const S = Math.min(3, Math.sqrt(14e6 / (W * H)));          // nitidez máxima sin pasarse del límite del móvil
  const cv = document.createElement('canvas'); cv.width = Math.round(W * S); cv.height = Math.round(H * S);
  const c = cv.getContext('2d'); c.scale(S, S); c.textBaseline = 'middle';
  const xs = []; let x0 = PAD; cols.forEach(k => { xs.push(x0); x0 += k.w; });

  c.fillStyle = '#fff'; c.fillRect(0, 0, W, H);
  c.fillStyle = '#4c1d95'; c.fillRect(0, 0, W, 120);
  c.fillStyle = '#fff'; c.textAlign = 'left';
  c.font = 'bold 54px Arial'; c.fillText('Ranking ' + cat + ' · Temporada ' + T.etiqueta, PAD, 45);
  c.font = '28px Arial'; c.fillText('Liga de viernes UNI · solo cuentan las ' + N + ' mejores jornadas de cada jugador', PAD, 93);

  c.fillStyle = '#ede9fe'; c.fillRect(PAD, top, W - 2 * PAD, headH);
  cols.forEach(function (k, i) {
    c.fillStyle = '#4c1d95'; c.textAlign = k.a; c.font = 'bold 26px Arial';
    const cx = k.a === 'left' ? xs[i] + 16 : xs[i] + k.w / 2;
    if (k.s) { c.fillText(k.h, cx, top + 30); c.font = '20px Arial'; c.fillStyle = '#6b6486'; c.fillText(k.s, cx, top + 62); }
    else c.fillText(k.h, cx, top + headH / 2);
  });

  C.jugadores.forEach(function (j, r) {
    const y = top + headH + r * rowH;
    const cuentan = j.pts.map((p, i) => ({ p: p, i: i })).filter(o => o.p !== null).sort((a, b) => (b.p - a.p) || (a.i - b.i)).slice(0, N).map(o => o.i);
    if (r % 2) { c.fillStyle = '#f7f5ff'; c.fillRect(PAD, y, W - 2 * PAD, rowH); }
    c.fillStyle = '#e5e0f5'; c.fillRect(PAD, y + rowH - 1, W - 2 * PAD, 1);
    fila(j).forEach(function (val, ci) {
      const k = cols[ci], s = String(val), cx = k.a === 'left' ? xs[ci] + 16 : xs[ci] + k.w / 2;
      const cuenta = k.jor !== undefined && cuentan.indexOf(k.jor) !== -1;
      if (cuenta) { c.fillStyle = '#e3f7ea'; c.fillRect(xs[ci] + 4, y + 4, k.w - 8, rowH - 8); }
      if (ci === 0 && j.pos <= 3) { c.fillStyle = ['#f5c542', '#cfd4dc', '#d9965b'][j.pos - 1]; c.beginPath(); c.arc(cx, y + rowH / 2, 22, 0, 7); c.fill(); }
      const bold = ci === 0 || (ci === 1 && j.pos <= 3) || k.tot || cuenta;
      c.textAlign = k.a;
      if (k.a === 'left') fitTexto(c, s, k.w - 28, 27, bold ? 'bold' : 'normal'); else c.font = (bold ? 'bold ' : '') + '27px Arial';
      c.fillStyle = k.tot ? '#4c1d95' : cuenta ? '#15803d' : (ci > 2 && s.charAt(0) === '-') ? '#dc2626' : s === '–' ? '#a39bbd' : '#1f1a33';
      c.fillText(s, cx, y + rowH / 2 + 2);
    });
  });
  c.fillStyle = '#c4b5fd';
  cols.forEach(function (k, i) { if (k.sep) c.fillRect(xs[i], top, 1.5, headH + n * rowH); });
  c.fillRect(PAD, top + headH + n * rowH, W - 2 * PAD, 1.5);

  c.fillStyle = '#6b6486'; c.font = '22px Arial'; c.textAlign = 'left';
  c.fillText('Verde: jornadas que cuentan · PG/PP: partidos ganados/perdidos · PF/PC: puntos a favor/en contra · Dif: diferencia', PAD, H - 68);
  c.fillText('Desempates: victorias, dif. de partidos, dif. de puntos, puntos a favor', PAD, H - 34);
  c.textAlign = 'right'; c.fillText('Actualizado: ' + datos.generado, W - PAD, H - 34);
  entregar(cv, formato, 'Ranking_' + slug(cat) + '_' + T.id, 842, 595);   // A4 horizontal
}
