function srv(nombre, args, ok, errorId, onFail) {
  let backend = null;
  if (typeof google !== 'undefined' && google.script && google.script.run) { backend = google.script.run; }
  else { if (!window._mockRunner) window._mockRunner = createMockRunner(); backend = window._mockRunner; }

  const runner = backend.withSuccessHandler(function(result) { if (ok) ok(result); })
    .withFailureHandler(function(error) {
      const message = (error && error.message) ? error.message : String(error || 'Error desconocido');
      const clean = message.replace(/^Error:\s*/, '');
      if (errorId && $(errorId)) { $(errorId).textContent = clean; }
      if (onFail) { onFail(clean); }
    });
  runner[nombre].apply(runner, [PIN].concat(args || []));
}

function crearBuscador(parent, placeholder, posIndex, callback) {
  const wrap = document.createElement('div'); wrap.className = 'search';
  const input = document.createElement('input'); input.placeholder = placeholder; input.autocomplete = 'off';
  const suggestions = document.createElement('div'); suggestions.className = 'suggest'; wrap.appendChild(input); wrap.appendChild(suggestions); parent.appendChild(wrap);
  function paint() {
    const q = normaliza(input.value.trim());
    suggestions.innerHTML = '';
    const reqSexo = sexoRequerido(CATEGORIA, posIndex || 0);

    const candidatos = SOCIOS.filter(function(socio) {
      const sSexo = normalizarSexo(socio.sexo);
      if (reqSexo && sSexo && sSexo !== reqSexo) return false;
      return true;
    });
    const resultados = candidatos.filter(function(socio) { if (!q) return true; return normaliza(getNombreCompleto(socio)).includes(q); }).slice(0, 12);
    if (!resultados.length) { suggestions.innerHTML = '<div>Sin resultados</div>'; return; }
    resultados.forEach(function(socio) {
      const option = document.createElement('div'); const nombreCompleto = getNombreCompleto(socio); const sexTag = normalizarSexo(socio.sexo) || 'General';
      option.innerHTML = '<span>' + esc(nombreCompleto) + '</span><span class="tag">' + esc(socio.categoria || '') + ' · ' + esc(sexTag) + '</span>';
      option.onclick = function(e) { e.stopPropagation(); input.value = nombreCompleto; suggestions.innerHTML = ''; callback(socio); }; suggestions.appendChild(option);
    });
  }
  input.addEventListener('input', paint); input.addEventListener('focus', paint); input.addEventListener('click', paint); return input;
}
document.addEventListener('click', function(e) { if (!e.target.closest('.search')) { document.querySelectorAll('.suggest').forEach(s => s.innerHTML = ''); } });

function quitarParticipante(idPart) {
  const idx = DETALLE.participantes.findIndex(p => p.id === idPart);
  const quitado = idx !== -1 ? DETALLE.participantes[idx] : null;
  if (idx !== -1) {
    DETALLE.participantes.splice(idx, 1);
    renderParticipantList();
    renderSystemConfig();
  }
  srv('quitarInscrito', [JORNADA.id, CATEGORIA, idPart], null, 'configErr', function(msj) {
    if (quitado) {
      DETALLE.participantes.splice(idx, 0, quitado);
      renderParticipantList();
      renderSystemConfig();
    }
    $('configErr').textContent = msj;
  });
}

function potenciaDosSiguiente(numero) { let size = 2; while (size < numero) { size *= 2; } return size; }
function randomSeedArray(ids, size) { const shuffled = ids.slice().sort(() => Math.random() - 0.5); const result = Array(size).fill(''); let cursor = 0; let byes = size - shuffled.length; for (let i = 0; i < size / 2; i++) { if (cursor >= shuffled.length) break; result[i * 2] = shuffled[cursor++]; if (byes > 0) { byes--; continue; } if (cursor < shuffled.length) { result[i * 2 + 1] = shuffled[cursor++]; } } return result; }

function abrirPrepararFinal() {
  $('finalErr').textContent = '';

    clasificados = clasificados || [];

    FINAL_SEED_IDS = clasificados.map(function(x) { return String(x.id); });
    FINAL_SEED_SIZE = potenciaDosSiguiente(FINAL_SEED_IDS.length);
    const byes = FINAL_SEED_SIZE - FINAL_SEED_IDS.length;
    $('finalInfo').innerHTML = '<b>' + FINAL_SEED_IDS.length + ' clasificados</b> · cuadro R' + FINAL_SEED_SIZE + (byes ? ' · ' + byes + ' BYE' + (byes === 1 ? '' : 's') : '');

function recogerSets() {
  const sets = [];
  for (let i = 0; i < 3; i++) {
    const inputA = $('sa' + i);
    const inputB = $('sb' + i);
    const valA = inputA ? inputA.value.trim() : '';
    const valB = inputB ? inputB.value.trim() : '';
    
    if (valA === '' && valB === '') {
      if (i === 0) throw new Error('Introduce al menos el marcador del primer set.');
      continue;
    }
    
    if (valA === '' || valB === '') throw new Error('Completa el marcador del set ' + (i + 1) + '.');

    const a = Number(valA);
    const b = Number(valB);
    if (!Number.isInteger(a) || !Number.isInteger(b) || a < 0 || b < 0) {
      throw new Error('Los marcadores deben ser números enteros válidos.');
    }
    if (a === b) throw new Error('Un set no puede terminar en empate.');
    sets.push([a, b]);
  }
  return sets;
}

function slug(s) { return String(s).replace(/[^\w-]+/g, '_'); }

function fitTexto(c, txt, maxW, fs, peso) {
  let s = fs;
  do { c.font = peso + ' ' + s + 'px Arial, Helvetica, sans-serif'; if (c.measureText(txt).width <= maxW) break; s -= 1; } while (s > 10);
  return s;
}

// Mete un JPEG en un PDF de una página (pw x ph en puntos), centrado y ajustado.
function crearPdf(jpegUrl, iw, ih, pw, ph) {
  const bin = atob(jpegUrl.split(',')[1]), jpg = new Uint8Array(bin.length);
  for (let i = 0; i < bin.length; i++) jpg[i] = bin.charCodeAt(i);
  const k = Math.min(pw / iw, ph / ih), w = (iw * k).toFixed(2), h = (ih * k).toFixed(2);
  const x = ((pw - iw * k) / 2).toFixed(2), y = ((ph - ih * k) / 2).toFixed(2);
  const enc = new TextEncoder(), parts = [], offs = []; let len = 0;
  const add = d => { const u = typeof d === 'string' ? enc.encode(d) : d; parts.push(u); len += u.length; };
  const obj = (n, head, stream) => { offs[n] = len; add(n + ' 0 obj\n' + head + '\n'); if (stream) { add('stream\n'); add(stream); add('\nendstream\n'); } add('endobj\n'); };
  const cont = 'q ' + w + ' 0 0 ' + h + ' ' + x + ' ' + y + ' cm /Im0 Do Q';
  add('%PDF-1.4\n');
  obj(1, '<< /Type /Catalog /Pages 2 0 R >>');
  obj(2, '<< /Type /Pages /Kids [3 0 R] /Count 1 >>');
  obj(3, '<< /Type /Page /Parent 2 0 R /MediaBox [0 0 ' + pw + ' ' + ph + '] /Resources << /XObject << /Im0 4 0 R >> >> /Contents 5 0 R >>');
  obj(4, '<< /Type /XObject /Subtype /Image /Width ' + iw + ' /Height ' + ih + ' /ColorSpace /DeviceRGB /BitsPerComponent 8 /Filter /DCTDecode /Length ' + jpg.length + ' >>', jpg);
  obj(5, '<< /Length ' + cont.length + ' >>', cont);
  const xr = len; add('xref\n0 6\n0000000000 65535 f \n');
  for (let n = 1; n <= 5; n++) add(String(offs[n]).padStart(10, '0') + ' 00000 n \n');
  add('trailer\n<< /Size 6 /Root 1 0 R >>\nstartxref\n' + xr + '\n%%EOF');
  return new Blob(parts, { type: 'application/pdf' });
}

function descargar(blob, nombre) {
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob); a.download = nombre;
  document.body.appendChild(a); a.click();
  setTimeout(function () { URL.revokeObjectURL(a.href); a.remove(); }, 1500);
}

function entregar(cv, formato, nombre, pw, ph) {
  try {
    if (formato === 'jpg') cv.toBlob(function (b) { descargar(b, nombre + '.jpg'); }, 'image/jpeg', 0.95);
    else descargar(crearPdf(cv.toDataURL('image/jpeg', 0.95), cv.width, cv.height, pw, ph), nombre + '.pdf');
  } catch (e) { alert('No se ha podido generar el archivo: ' + e.message); }
}

   LLAVE PARA IMPRIMIR: blanco y negro, sin relleno, con huecos para escribir a mano
   ================================================================ */
function exportarLlave(formato) {
  const P = (DETALLE.partidos || []).filter(p => p.ronda !== 'Fase de grupos');
  const rondas = ['Dieciseisavos', 'Octavos', 'Cuartos', 'Semifinal', 'Final'].filter(r => P.some(p => p.ronda === r));
  if (!rondas.length) return;
  const n0 = P.filter(p => p.ronda === rondas[0]).length * 2, grande = n0 > 32;   // >32 jugadores -> A3
  const PW = grande ? 4961 : 3508, PH = grande ? 3508 : 2480, f = PW / 3508, m = 150 * f, top = 330 * f;
  const cv = document.createElement('canvas'); cv.width = PW; cv.height = PH;
  const c = cv.getContext('2d');
  c.fillStyle = '#fff'; c.fillRect(0, 0, PW, PH);
  c.fillStyle = c.strokeStyle = '#000'; c.lineWidth = 3 * f; c.textAlign = 'left';
  c.font = 'bold ' + 90 * f + 'px Arial'; c.fillText('Cuadro ' + CATEGORIA, m, 150 * f);
  c.font = 46 * f + 'px Arial'; c.fillText(fechaLarga(JORNADA.fecha) + ' · ' + JORNADA.modalidad, m, 225 * f);
  c.font = 36 * f + 'px Arial'; c.fillText('Liga de los viernes UNI · Temporada 2026/27 · ' + textoFormato(), m, PH - 60 * f);

  const R = rondas.length, gk = 0.16;
  const W = Math.min((PW - 2 * m) / (R + 1 + R * gk), 720 * f), gap = W * gk;
  const slot = Math.min((PH - top - m) / n0, 190 * f), yTop = top + ((PH - top - m) - slot * n0) / 2;
  const fs = Math.min(slot * 0.42, 60 * f);
  // y de cada línea de entrada: la primera ronda es fija; las siguientes salen del centro de su partido anterior
  const eY = (ri, j) => ri === 0 ? yTop + (j + 0.5) * slot : cen(ri - 1, j + 1);
  const cen = (ri, k) => (eY(ri, 2 * k - 2) + eY(ri, 2 * k - 1)) / 2;

  rondas.forEach(function (ronda, ri) {
    const x = m + ri * (W + gap);
    c.font = 'bold ' + fs * 1.05 + 'px Arial'; c.textAlign = 'center'; c.fillText(ronda.toUpperCase(), x + W / 2, yTop - slot * 0.15); c.textAlign = 'left';
    P.filter(p => p.ronda === ronda).forEach(function (p) {
      const k = p.orden;
      [['a', 2 * k - 2], ['b', 2 * k - 1]].forEach(function (par) {
        const lado = par[0], y = eY(ri, par[1]), nom = (p[lado] && p[lado].nombre) || '';
        c.beginPath(); c.moveTo(x, y); c.lineTo(x + W, y); c.stroke();
        const marc = (p.sets || []).map(s => s[lado === 'a' ? 0 : 1]).join('  ');
        const txt = nom || (p.estado === 'Bye' ? 'bye' : '');
        const est = txt === 'bye' ? 'italic' : (p.ganador && p.ganador.toLowerCase() === lado ? 'bold' : 'normal');
        c.font = fs * 0.8 + 'px Arial'; const sw = marc ? c.measureText(marc).width + 24 * f : 0;
        if (marc) { c.textAlign = 'right'; c.fillText(marc, x + W - 8 * f, y - fs * 0.2); c.textAlign = 'left'; }
        if (txt) { fitTexto(c, txt, W - 16 * f - sw, fs, est); c.fillText(txt, x + 8 * f, y - fs * 0.2); }
      });
      const yA = eY(ri, 2 * k - 2), yB = eY(ri, 2 * k - 1), yc = (yA + yB) / 2;
      c.beginPath(); c.moveTo(x + W, yA); c.lineTo(x + W, yB); c.moveTo(x + W, yc); c.lineTo(x + W + gap, yc); c.stroke();
      if (ri === R - 1) {   // línea del campeón
        const xc = x + W + gap; c.beginPath(); c.moveTo(xc, yc); c.lineTo(xc + W, yc); c.stroke();
        c.font = 'bold ' + fs * 0.8 + 'px Arial'; c.fillText('CAMPEÓN/A', xc + 8 * f, yc - slot * 0.6);
        const g = p.ganador && p[p.ganador.toLowerCase()];
        if (g && g.nombre) { fitTexto(c, g.nombre, W - 16 * f, fs, 'bold'); c.fillText(g.nombre, xc + 8 * f, yc - fs * 0.2); }
      }
    });
  });
  entregar(cv, formato, 'Llave_' + slug(CATEGORIA) + '_' + JORNADA.fecha, grande ? 1191 : 842, grande ? 842 : 595);
}
